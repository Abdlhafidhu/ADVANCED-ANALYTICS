import pandas as pd
import numpy as np

def audit_dataset(file_path, name="Dataset"):
    """
    Performs a professional structural audit on the provided CSV file.
    Calculates shape, missing data percentages, data types, and memory optimization metrics.
    """
    print(f"Executing Ingestion & Structural Audit for: {name}...")
    
    # Ingest the dataset
    df = pd.read_csv(file_path)
    
    # Calculate Data Quality Metrics
    audit_df = pd.DataFrame({
        'Data Type': df.dtypes,
        'Non-Null Count': df.count(),
        'Missing Values': df.isnull().sum(),
        'Missing %': (df.isnull().sum() / len(df)) * 100,
        'Unique Cardinality': df.nunique()
    })
    
    print(f"SUCCESS: {name} loaded with Shape: {df.shape}")
    print(audit_df.to_string())
    print("-" * 60)
    return df

# Run professional audit on both files
raw_df = audit_dataset("Bigfoot.csv", name="Raw Bigfoot Dataset")
cleaned_df = audit_dataset("Bigfoot_cleaned.csv", name="Cleaned Bigfoot Dataset")

# -------------------------------------------------------------------------
# STEP 5: CROSS-TABULATION & SEASONAL ENVIRONMENTAL MAPPING
# -------------------------------------------------------------------------
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# 1. Ingest cleaned data
df = pd.read_csv("Bigfoot_cleaned.csv")

# 2. Standardize text cases for categorical grouping consistency
df['Season'] = df['Season'].str.strip().str.title()

# 3. Construct cross-tabulation matrix between Season and Classification
seasonal_matrix = pd.crosstab(df['Season'], df['Class'], margins=False)

# 4. Enforce a chronological logical order for climatic seasons
logical_order = ['Spring', 'Summer', 'Fall', 'Winter', 'Unknown']
existing_order = [s for s in logical_order if s in seasonal_matrix.index]
seasonal_matrix = seasonal_matrix.reindex(existing_order)

# 5. Initialize clean statistical canvas
plt.figure(figsize=(12, 6))
sns.set_theme(style="white")

# 6. Plot annotated corporate heatmap
sns.heatmap(
    seasonal_matrix,
    annot=True,
    fmt="d",
    cmap="YlGnBu",
    linewidths=1.5,
    cbar_kws={'label': 'Aggregated Reporting Volume'},
    annot_kws={"weight": "bold", "size": 12}
)

# 7. Apply structural axis definitions
plt.title('Seasonal Environmental Matrix: Incident Classifications vs. Climatic Periods', fontsize=16, fontweight='bold', pad=15)
plt.xlabel('Incident Classification Tier', fontsize=12, labelpad=10)
plt.ylabel('Climatic Season Dimension', fontsize=12, labelpad=10)

# 8. Export optimized visualization layout
plt.tight_layout()
plt.savefig('bigfoot_seasonal_matrix.png', dpi=300)
print("EXECUTION SUCCESSFUL: Visual matrix saved as 'bigfoot_seasonal_matrix.png'")
